#!/usr/bin/env python3
"""Saved-only lifecycle audit of the first restoration-controls-v2 study.

Read JSON, source bytes and artifact bytes only. No subject imports, reference
generation, image decoding, optimizer, model, point construction or new census.
Reported scientific errors are checked against unchanged targets; their forward,
boundary, rank and physical accuracy are not independently established here.
"""

from __future__ import annotations

import argparse
import collections
import hashlib
import json
import math
import stat
from pathlib import Path

HEAD = "e677039274707f8c19e3efaa005f69d74b393868"
REPORT_SHA = "4eadc2057ded427a4727cb2e14cae0aac153907fd511252423529dcd612fc982"
FREEZE_SHA = "7d244b9b197217dd7b06718bdde69769cf3f163489192b267e211d6ecaab726b"
TERMINAL_SHA = "a9a6870357a01c064f70c8eb92ce89cadd41c23f2fd743c6d2cf1f37999c123c"
CLOSURE_SHA = "33ce7a777b732e6970092c119bb18f7b72e234a2b15ca37c92ba2c9274136576"
ORIGINAL_PLAN_SHA = "737bb8badf15a6bd1438ece67b9ad03f96e65440989bbc8bc2110edc8a497ad0"
FORMAT = "finite-acquisition-recovery-restoration-controls-v2-complete-report"


def decode(raw):
    def number(text):
        value = float(text)
        if not math.isfinite(value):
            raise ValueError("nonfinite JSON number")
        return value

    def constant(text):
        raise ValueError("nonfinite JSON token: " + text)

    def pairs(items):
        value = {}
        for key, item in items:
            if key in value:
                raise ValueError("duplicate JSON key: " + key)
            value[key] = item
        return value

    return json.loads(raw, parse_float=number, parse_constant=constant, object_pairs_hook=pairs)


def binding(path):
    digest, size = hashlib.sha256(), 0
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
            size += len(block)
    return {"sha256": digest.hexdigest(), "bytes": size}


def safe(root, name):
    relative = Path(name)
    if relative.is_absolute() or ".." in relative.parts:
        raise ValueError("unconfined artifact/source path")
    current = root
    for part in relative.parts:
        current = current / part
        if current.is_symlink():
            raise ValueError("symlink in artifact/source path")
    if not current.resolve().is_relative_to(root.resolve()):
        raise ValueError("path escaped its root")
    if not stat.S_ISREG(current.stat().st_mode):
        raise ValueError("not a regular file")
    return current


class Audit:
    def __init__(self):
        self.checks = 0
        self.failures = []
        self.rows = []

    def check(self, condition, context):
        self.checks += 1
        if not condition:
            self.failures.append(context)

    def bound(self, path, expected, context):
        try:
            actual = binding(path)
            self.check(actual == {key: expected[key] for key in ("sha256", "bytes")}, context)
            return actual
        except (OSError, ValueError, KeyError) as error:
            self.check(False, context + ": " + f"{type(error).__name__}: {error}")
            return None


def unquarantine(value):
    value = dict(value)
    if "all_targets_before_source_gate" in value:
        value["all_targets"] = value["all_targets_before_source_gate"]
    elif "all_targets_before_protocol_gate" in value:
        value["all_targets"] = value["all_targets_before_protocol_gate"]
    for key in (
        "all_targets_before_source_gate",
        "all_targets_before_protocol_gate",
        "source_gate_interpretation_withheld",
        "protocol_gate_interpretation_withheld",
        "interpretation_withheld",
    ):
        value.pop(key, None)
    if isinstance(value.get("evaluation"), dict):
        value["evaluation"] = unquarantine(value["evaluation"])
    return value


def audit_row(audit, row, case, run, plan, artifact_names, quarantined):
    name = case["name"]
    audit.check(row["case"] == case, name + "/case_identity")
    audit.check(type(row.get("all_targets")) is bool, name + "/native_target_flag")
    record_name = name + "/record.json"
    if record_name in artifact_names:
        first = decode(safe(run, record_name).read_bytes())
        audit.check(unquarantine(row) == first, name + "/first_row_retained")
    else:
        audit.check(
            row["status"] in ("exception", "not_started"), name + "/missing_record_disposition"
        )
    evaluation = row.get("evaluation") or {}
    flags = evaluation.get("targets", {})
    if evaluation.get("status") == "evaluated":
        for key, error in evaluation["errors"].items():
            audit.check(
                flags.get(key) is (error is not None and error <= plan["targets"][key]),
                name + "/target/" + key,
            )
        audit.check(
            evaluation["all_targets"] is (all(flags.values()) and not quarantined),
            name + "/nested_all_targets",
        )
        reserved = evaluation["conditional_reserved_residuals"]
        maximum = max((abs(value) for value in reserved), default=None)
        audit.check(
            maximum == evaluation["errors"]["conditional_query_max_residual_linear_level"],
            name + "/reserved_residual_maximum",
        )
    reference = row.get("reference", {})
    if reference:
        audit.check(
            reference["passed"]
            is (
                reference["max_area_order_delta"] <= plan["reference"]["area_max_delta"]
                and reference["encoded_order_codes_equal"] is not False
            ),
            name + "/reference_gate",
        )
    expected = (
        row["status"] == "completed"
        and reference.get("passed") is True
        and evaluation.get("all_targets") is True
        and not quarantined
    )
    audit.check(row["all_targets"] is expected, name + "/row_all_targets")
    if row["status"] != "completed":
        audit.check(row["all_targets"] is False, name + "/noncompleted_not_success")
    fit_name = name + "/fit.json"
    fit_summary = None
    if fit_name in artifact_names:
        fitted = decode(safe(run, fit_name).read_bytes())
        audit.check(row.get("fit_attempted") is True, name + "/fit_artifact_attempted")
        fit_inputs = decode(safe(run, name + "/observed-fit-input.json").read_bytes())
        init = decode(safe(run, name + "/initialization.json").read_bytes())
        audit.check(
            fit_inputs["preparation_input_sha256"] == init["input_sha256"],
            name + "/initializer_input_binding",
        )
        audit.check(fit_inputs["mode"] == case["mode"] == fitted["mode"], name + "/mode")
        all_queries = init["queries"]
        training = [query for query in all_queries if (query[0] + 2 * query[1]) % 5 != 0]
        if case["adverse"] == "five":
            training = training[:5]
        selected = {tuple(query) for query in training}
        reserved = [query for query in all_queries if tuple(query) not in selected]
        audit.check(fit_inputs["queries"] == training, name + "/training_mask_and_override")
        audit.check(
            fit_inputs["reserved_queries"] == reserved, name + "/reserved_mask_and_override"
        )
        limits = ("max_evaluations", "max_cell_work", "max_stage_work")
        controls = fit_inputs["controls"]
        audit.check(
            controls["max_iterations"] == plan["optimizer"]["max_iterations"],
            name + "/unchanged_iterations",
        )
        audit.check(
            controls["level_bounds"] == plan["optimizer"]["level_bounds"],
            name + "/unchanged_level_bounds",
        )
        expected_limits = (
            [1, 169, 100000]
            if case["adverse"] == "budget"
            else [plan["optimizer"][key] for key in limits]
        )
        audit.check([controls[key] for key in limits] == expected_limits, name + "/unchanged_caps")
        audit.check(fitted["limits"] == expected_limits, name + "/fit_caps")
        for key, cap in zip(
            ("evaluations", "cell_work", "stage_work"), expected_limits, strict=True
        ):
            used = fitted[key]
            audit.check(type(used) is int and 0 <= used <= cap, name + "/charged_total/" + key)
        best = fitted.get("best")
        if best is not None and evaluation.get("status") == "evaluated":
            audit.check(best["parameters"] == evaluation["parameters"], name + "/selected_geometry")
            audit.check(best["levels"] == evaluation["levels"], name + "/selected_levels")
            completion = (
                not fitted["work_exhausted"]
                and not fitted["protocol_failure"]
                and not fitted["completion_withheld"]
                and any(
                    search["best_candidate_index"] == best["index"]
                    and search["status"] in ("projected_gradient_resolution", "poll_resolution")
                    for search in fitted["searches"]
                )
            )
            audit.check(
                flags["bounded_local_search_completion"] is completion, name + "/completion_link"
            )
            if case["mode"] == "smooth":
                audit.check(
                    flags["geometry_rank_conservative_when_smooth"]
                    is (
                        best["geometry_rank_conservative"]
                        == plan["targets"]["geometry_rank_conservative_when_smooth"]
                    ),
                    name + "/reported_smooth_rank_gate",
                )
            fit_summary = {
                "cost": best["cost"],
                "tied": fitted["tied"],
                "work_exhausted": fitted["work_exhausted"],
                "completion_withheld": fitted["completion_withheld"],
                "search_statuses": [search["status"] for search in fitted["searches"]],
            }
        del fitted
    else:
        audit.check(row.get("fit_attempted", False) is False, name + "/no_invented_fit")
    audit.rows.append(
        {
            "case": name,
            "family": case["family"],
            "status": row["status"],
            "fit_attempted": row.get("fit_attempted", False),
            "all_targets": row["all_targets"],
            "failed_targets": [key for key, value in flags.items() if not value],
            "reported_errors": evaluation.get("errors"),
            "selected_fit": fit_summary,
        }
    )


def main(argv=None):
    parser = argparse.ArgumentParser(description=__doc__)
    for name in ("run-dir", "freeze", "launch", "terminal", "controls", "log", "output"):
        parser.add_argument("--" + name, required=True, type=Path)
    parser.add_argument("--expected-report-sha256", required=True)
    parser.add_argument("--source-root", type=Path)
    args = parser.parse_args(argv)
    output = args.output.absolute()
    if (
        any(path.is_symlink() for path in (output, *output.parents))
        or output.exists()
        or not output.parent.is_dir()
    ):
        parser.error("a fresh nonsymlink output with an existing parent is required")
    audit = Audit()
    receipt = {
        "format": "restoration-v2-ux-saved-lifecycle-audit-v1",
        "source_imports": 0,
        "model_reference_optimizer_calls": 0,
        "new_points_or_census": 0,
        "summary": None,
    }
    paths = {key: getattr(args, key) for key in ("freeze", "launch", "terminal", "controls", "log")}
    paths["report"] = args.run_dir / "report.json"
    first_bindings = {}
    try:
        for key, path in paths.items():
            first_bindings[key] = binding(path)
        audit.check(args.expected_report_sha256 == REPORT_SHA, "admitted_canonical_report")
        audit.check(
            first_bindings["report"]["sha256"] == args.expected_report_sha256, "report_input_hash"
        )
        audit.check(first_bindings["freeze"]["sha256"] == FREEZE_SHA, "outer_freeze_input_hash")
        audit.check(first_bindings["terminal"]["sha256"] == TERMINAL_SHA, "terminal_input_hash")
        report = decode(paths["report"].read_bytes())
        freeze = decode(args.freeze.read_bytes())
        terminal = decode(args.terminal.read_bytes())
        launch = decode(args.launch.read_bytes())
        controls = decode(args.controls.read_bytes())
        receipt["input_bindings"] = first_bindings
        audit.check(report["format"] == FORMAT, "report_format")
        audit.check(
            report["source_head"] == report["source_head_after"] == freeze["head"] == HEAD,
            "source_head",
        )
        frozen, modes = freeze["governing_inputs"], freeze["git_modes"]
        rows = [[path, modes[path], frozen[path]] for path in sorted(frozen)]
        closure = hashlib.sha256(
            json.dumps(rows, separators=(",", ":"), ensure_ascii=True).encode("ascii")
        ).hexdigest()
        audit.check(
            len(frozen) == len(modes) == 131 and set(frozen) == set(modes), "131_frozen_sources"
        )
        audit.check(closure == freeze["closure_sha256"] == CLOSURE_SHA, "source_closure")
        for phase in ("source_before", "source_after"):
            snapshot = report[phase]
            audit.check(
                snapshot["complete"] is True and snapshot["errors"] == [], phase + "/complete"
            )
            audit.check(
                snapshot["sha256"] == frozen and set(snapshot["records"]) == set(frozen),
                phase + "/identities",
            )
            for path, digest in frozen.items():
                record = snapshot["records"][path]
                audit.check(
                    record["sha256"] == digest
                    and record["git_mode"] == modes[path]
                    and record["error"] is None,
                    phase + "/" + path,
                )
        source_freeze = decode(safe(args.run_dir, "source-freeze.json").read_bytes())
        audit.check(
            source_freeze["head"] == HEAD and source_freeze["sha256"] == frozen,
            "run_freeze_binding",
        )
        plan = report["original_plan"]
        audit.check(plan == source_freeze["plan"], "original_plan_snapshot")
        audit.check(
            report["continuation_admission"] == source_freeze["continuation_admission"],
            "admission_snapshot",
        )
        audit.check(
            report["admission_input_sha256"]
            == frozen["studies/finite-acquisition-recovery-restoration-controls-v2.json"],
            "consumed_admission_source_binding",
        )
        population = plan["population"]
        audit.check(
            len(population) == 24 and len({case["name"] for case in population}) == 24,
            "24_unique_cases",
        )
        audit.check(
            collections.Counter(case["family"] for case in population)
            == {"matched": 12, "adverse": 12},
            "12_matched_12_adverse",
        )
        audit.check(
            [row["case"] for row in report["rows"]] == population,
            "full_population_order_and_fields",
        )
        artifacts = report["artifacts"]
        actual = {
            str(path.relative_to(args.run_dir))
            for path in args.run_dir.rglob("*")
            if path.is_file()
        }
        audit.check(actual == set(artifacts) | {"report.json"}, "complete_artifact_inventory")
        for name, expected in artifacts.items():
            try:
                audit.bound(safe(args.run_dir, name), expected, "artifact/" + name)
            except (OSError, ValueError) as error:
                audit.check(False, "artifact/" + name + ": " + str(error))
        quarantined = report["source_unchanged"] is not True or bool(report["protocol_errors"])
        for row, case in zip(report["rows"], population, strict=True):
            try:
                audit_row(audit, row, case, args.run_dir, plan, set(artifacts), quarantined)
            except (OSError, ValueError, TypeError, KeyError, OverflowError) as error:
                audit.check(
                    False, case["name"] + "/audit_exception: " + f"{type(error).__name__}: {error}"
                )
        matched = sum(
            row["all_targets"] is True
            for row in report["rows"]
            if row["case"]["family"] == "matched"
        )
        adverse = sum(
            row["all_targets"] is True
            for row in report["rows"]
            if row["case"]["family"] == "adverse"
        )
        passed = not quarantined and len(report["rows"]) == 24 and matched == 12
        audit.check(
            report["matched_passes"] == matched and report["matched_total"] == 12, "matched_count"
        )
        audit.check(report["adverse_all_target_passes"] == adverse, "adverse_count_separate")
        audit.check(report["passed"] is passed, "overall_pass_definition")
        expected_exit = 0 if passed else 1
        audit.check(
            terminal["native_session_id"] == 28887
            and terminal["attempt"] == 1
            and terminal["reruns"] == 0,
            "first_native_only",
        )
        audit.check(
            terminal["exit_code"] == expected_exit and terminal["passed"] is passed,
            "native_exit_reconciliation",
        )
        for key in ("report", "log", "launch", "freeze"):
            audit.check(
                all(
                    terminal[key][field] == first_bindings[key][field]
                    for field in ("sha256", "bytes")
                ),
                "terminal_binding/" + key,
            )
        for key in ("freeze", "controls_receipt"):
            source_key = "controls" if key == "controls_receipt" else key
            audit.check(
                all(
                    launch[key][field] == first_bindings[source_key][field]
                    for field in ("sha256", "bytes")
                ),
                "launch_binding/" + key,
            )
        audit.check(
            launch["attempt"] == 1 and launch["reruns"] == 0 and launch["source_head"] == HEAD,
            "launch_first_source",
        )
        audit.check(
            controls["exit_code"] == 0
            and controls["passed"] == 1169
            and controls["subtests_passed"] == 33
            and controls["deselected_exact_malformed_old_node"] == 1,
            "selected_controls_scope",
        )
        events = [decode(line) for line in args.log.read_bytes().splitlines() if line.strip()]
        audit.check(
            events[:-1]
            == [
                {
                    "name": row["case"]["name"],
                    "status": row["status"],
                    "all_targets": row["all_targets"],
                }
                for row in report["rows"]
            ],
            "native_log_all24_order",
        )
        audit.check(
            events[-1]
            == {key: report[key] for key in ("matched_passes", "matched_total", "passed")},
            "native_log_terminal_summary",
        )
        if args.source_root is not None:
            for path, expected in frozen.items():
                current = safe(args.source_root, path)
                audit.check(binding(current)["sha256"] == expected, "local_source_hash/" + path)
                mode = "100755" if current.stat().st_mode & 0o111 else "100644"
                audit.check(mode == modes[path], "local_source_mode/" + path)
            audit.check(
                binding(safe(args.source_root, "studies/finite-acquisition-recovery-v1.json"))[
                    "sha256"
                ]
                == ORIGINAL_PLAN_SHA,
                "original_plan_authority",
            )
            audit.check(
                decode(
                    safe(
                        args.source_root, "studies/finite-acquisition-recovery-v1.json"
                    ).read_bytes()
                )
                == plan,
                "all_original_plan_values_unchanged",
            )
            for key in ("prior_saved_bindings_retained", "prior_v1_failed_test_bindings_retained"):
                for name, expected in freeze[key].items():
                    audit.bound(
                        safe(args.source_root, expected.get("path", name)),
                        expected,
                        key + "/" + name,
                    )
        receipt["summary"] = {
            "rows": len(report["rows"]),
            "status_counts": dict(collections.Counter(row["status"] for row in report["rows"])),
            "matched_passes": matched,
            "matched_total": 12,
            "adverse_all_target_passes": adverse,
            "study_exit_code": terminal["exit_code"],
            "study_passed": passed,
            "source_records": len(frozen),
            "artifacts": len(artifacts),
            "row_details": audit.rows,
        }
    except (OSError, ValueError, TypeError, KeyError, OverflowError, IndexError) as error:
        audit.check(False, "top_level_audit_exception: " + f"{type(error).__name__}: {error}")
    for key, path in paths.items():
        if key in first_bindings:
            audit.bound(path, first_bindings[key], "audit_inputs_unchanged/" + key)
    receipt.update(
        checks=audit.checks,
        failures=audit.failures,
        passed=not audit.failures,
        auditor=binding(Path(__file__)),
        local_source_files_read=args.source_root is not None,
        exclusions=[
            "independent forward/reference/area/boundary/rank/stopping calculus",
            "rational certificate and correction derivation",
            "held-out statistical independence",
            "unknown-camera or physical accuracy",
        ],
    )
    with output.open("x") as stream:
        json.dump(receipt, stream, indent=2, allow_nan=False)
        stream.write("\n")
    print(
        json.dumps(
            {"checks": audit.checks, "failures": len(audit.failures), "passed": receipt["passed"]}
        )
    )
    return 0 if receipt["passed"] else 1


if __name__ == "__main__":
    raise SystemExit(main())
