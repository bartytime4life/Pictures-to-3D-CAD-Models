import hashlib
import json
import math
import struct
from pathlib import Path


ROOT = Path(__file__).resolve().parents[3]
FOLDER = Path(__file__).parent
RAW = ROOT / ".local/verification/finite-acquisition-defect-corrector-v1-first.json"
EXPECTED = "1738ab1ef1e5784dac9381c558a8f0947fbcf15b72208e432e7c8785f312c34d"
checks, errors = 0, []


def check(name, value):
    global checks
    checks += 1
    if value is not True:
        errors.append(name)


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def bind(path):
    digest, size = hashlib.sha256(), 0
    with path.open("rb") as stream:
        while block := stream.read(1024 * 1024):
            digest.update(block)
            size += len(block)
    return {"sha256": digest.hexdigest(), "bytes": size}


def load(path):
    raw = path.read_bytes()

    def number(text):
        value = float(text)
        if not math.isfinite(value):
            raise ValueError("nonfinite JSON number")
        return value

    def bad(text):
        raise ValueError("nonfinite token: " + text)

    def pairs(items):
        result = {}
        for key, value in items:
            if key in result:
                raise ValueError("duplicate JSON key")
            result[key] = value
        return result

    return json.loads(raw, parse_float=number, parse_constant=bad, object_pairs_hook=pairs), {"sha256": sha(raw), "bytes": len(raw)}


def bits(values):
    return b"".join(struct.pack("!d", value) for value in values)


def array(record):
    check("array rows", len(record["values"]) == record["shape"][0])
    check("array columns", all(len(row) == record["shape"][1] for row in record["values"]))
    values = [value for row in record["values"] for value in row]
    check("array finite", all(math.isfinite(value) for value in values))
    kind = {"<f4": "f", "<f8": "d", "|u1": "B"}[record["dtype"]]
    raw = struct.pack("<" + kind * len(values), *values)
    check("array raw hash", sha(raw) == record["sha256"])
    return raw
